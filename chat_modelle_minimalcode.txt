Hinweis:
Der bestehende Code verwendet einen frei einstellbaren baseUrl-Wert und ruft die OpenAI-kompatiblen Endpunkte direkt aus der HTML-Seite per fetch auf. Minimal brauchst du in der HTML-Seite:
- ein API-Key-Feld oder eine Variable apiKey
- baseUrl, z. B. "https://api.openai.com" oder die in der App eingestellte kompatible Base-URL
- fuer TTS ein <audio>-Element
- fuer Transkription ein File-Input oder eine MediaRecorder-Aufnahme als Blob

2) Text-to-Speech: openai/gpt-4o-mini-tts-2025-12-15
============================================================

Minimaler HTML-Teil:

<textarea id="ttsText">Hallo, das ist ein Test.</textarea>
<button onclick="generateSpeech()">Audio erzeugen</button>
<audio id="audioPlayer" controls></audio>

Minimaler JavaScript-Code:

<script>
async function generateSpeech() {
  const text = document.getElementById("ttsText").value.trim();
  const audioPlayer = document.getElementById("audioPlayer");

  if (!text) throw new Error("Bitte Text eingeben.");

  const response = await fetch(`${baseUrl.replace(/\/+$/, "")}/v1/audio/speech`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${apiKey}`
    },
    body: JSON.stringify({
      model: TTS_MODEL,
      input: text,
      voice: "alloy",
      response_format: "mp3",
      speed: 1
    })
  });

  if (!response.ok) {
    throw new Error(await response.text());
  }

  const blob = await response.blob();
  const audioUrl = URL.createObjectURL(blob);
  audioPlayer.src = audioUrl;
  await audioPlayer.play().catch(() => {});
}
</script>

Minimaler HTML-Teil:

Minimaler JavaScript-Code:

<script>
async function askGpt54Mini() {
  const userPrompt = document.getElementById("chatPrompt").value.trim();
  const output = document.getElementById("chatOutput");

  if (!userPrompt) throw new Error("Bitte Prompt eingeben.");

  const response = await fetch(`${baseUrl.replace(/\/+$/, "")}/v1/chat/completions`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${apiKey}`
    },
    body: JSON.stringify({
      model: GPT54_MINI_MODEL,
      messages: [
        { role: "system", content: "Du bist ein praeziser Lernassistent. Antworte klar und knapp auf Deutsch." },
        { role: "user", content: userPrompt }
      ],
      temperature: 0.3
    })
  });

  if (!response.ok) {
    throw new Error(await response.text());
  }

  const data = await response.json();
  output.textContent = data?.choices?.[0]?.message?.content ?? "";
}
</script>

4) Audio-to-Text / Transkription: openai/gpt-4o-transcribe
============================================================

Minimaler HTML-Teil mit Datei-Upload:

<input id="audioFile" type="file" accept="audio/*" />
<button onclick="transcribeAudioFile()">Audio transkribieren</button>
<pre id="transcriptOutput"></pre>

Minimaler JavaScript-Code:

<script>
async function transcribeAudioFile() {
  const fileInput = document.getElementById("audioFile");
  const output = document.getElementById("transcriptOutput");
  const file = fileInput.files?.[0];

  if (!file) throw new Error("Bitte Audiodatei auswaehlen.");

  const formData = new FormData();
  formData.append("model", TRANSCRIBE_MODEL);
  formData.append("file", file, file.name || "audio.webm");
  formData.append("prompt", "Transkribiere die medizinische Antwort praezise auf Deutsch.");

  const response = await fetch(`${baseUrl.replace(/\/+$/, "")}/v1/audio/transcriptions`, {
    method: "POST",
    headers: {
      "Authorization": `Bearer ${apiKey}`
    },
    body: formData
  });

  if (!response.ok) {
    throw new Error(await response.text());
  }

  const data = await response.json();
  output.textContent = String(data?.text ?? "").trim();
}
</script>

- Key-Auswahl fuer Transkription: index.html Zeile ca. 2894
- Request: index.html Zeile ca. 6430, fetch(`${baseUrl}/v1/audio/transcriptions`, ...)
